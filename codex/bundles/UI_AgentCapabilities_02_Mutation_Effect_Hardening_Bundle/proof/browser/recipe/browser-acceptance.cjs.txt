const fs = require('node:fs');
const path = require('node:path');
const assert = require('node:assert/strict');
const { chromium } = require(process.env.CAPA_PLAYWRIGHT);
const out = process.argv[2];
const control = 'http://127.0.0.1:17301';
const steps = [];
const call = async action => {
 const response = await fetch(control + '/fixture/' + action, { method:'POST' });
 assert.equal(response.ok,true,'fixture control: '+action);
};
const canonical = async () => (await fetch(control + '/fixture/state')).json();
async function poll(predicate,label) {
 const end=Date.now()+30000;
 while(Date.now()<end) {
  if(await predicate()) return;
  await new Promise(resolve=>setTimeout(resolve,100));
 }
 throw new Error('Timed out: '+label);
}
(async()=>{
 const browser=await chromium.launch({headless:true});
 const page=await browser.newPage({viewport:{width:1600,height:1000}});
 const errors=[];
 page.on('pageerror',error=>errors.push(error.message));
 const state=await canonical();
 const test=id=>page.getByTestId('agents-capability-'+id);
 const heading=page.locator('.agent-capabilities-panel__heading');
 const select=async name=>{
  await test('tree-agent').filter({hasText:'CAPA02 '+name}).click();
  await poll(async()=> (await heading.innerText()).includes('CAPA02 '+name),'selected '+name);
 };
 const shot=async name=>{
  await page.screenshot({path:path.join(out,name+'.jpeg'),type:'jpeg',quality:75});
  fs.writeFileSync(path.join(out,name+'-dom.txt'),await page.locator('body').innerText());
 };
 try {
  await page.goto('http://127.0.0.1:5273/agents?tab=capabilities&agentId='+state.alphaId);
  const proceed=page.getByRole('button',{name:'Continue',exact:true});
  await proceed.waitFor({timeout:90000});
  await proceed.click();
  await proceed.waitFor({state:'hidden'});
  await test('search').waitFor({timeout:90000});
  await poll(async()=> (await heading.innerText()).includes('CAPA02 Alpha'),'initial Alpha');
  await test('search').fill('CAPA02 Inline inspection');
  await test('type-filter').selectOption('Skill');
  await test('access-reason').fill('  Keep this raw rule  ');
  await test('access-server-key').fill('  local-raw  ');
  await poll(async()=>await test('card').count()===1,'single fixture capability');
  await shot('normal');
  await call('assignment/Rejected');
  await test('toggle').click();
  await poll(async()=> (await test('operation').innerText()).includes('rejected'),'rejection');
  assert.equal((await canonical()).assigned,false);
  assert.equal(await test('toggle').isEnabled(),true);
  await shot('rejected');
  steps.push('Rejected assignment leaves authoritative attachments unchanged and permits a corrected intent.');
  await call('hold/assignment');
  const beforePending=(await canonical()).assignmentCalls;
  await test('toggle').click();
  await poll(async()=> (await canonical()).held,'held assignment');
  assert.equal((await canonical()).assigned,false);
  await shot('assignment-pending');
  await select('Beta');
  await select('Alpha');
  assert.equal(await test('toggle').isDisabled(),true);
  assert.equal((await canonical()).assignmentCalls,beforePending+1);
  await call('release');
  await poll(async()=> (await canonical()).assigned,'single canonical assignment');
  await test('recover').waitFor();
  await test('recover').click();
  await poll(async()=>await test('toggle').isEnabled(),'reentry reconciliation');
  assert.equal((await canonical()).assignmentCalls,beforePending+1);
  assert.equal(await test('access-reason').inputValue(),'  Keep this raw rule  ');
  assert.equal(await test('access-server-key').inputValue(),'  local-raw  ');
  assert.equal(await test('search').inputValue(),'CAPA02 Inline inspection');
  await shot('reentry-reconciled');
  steps.push('Pending A -> B -> A retains one immutable attempt and original raw inputs; current A reconciliation performs no additional write.');
  await call('assignment/CommittedRefreshFailure');
  await test('toggle').click();
  await test('load-failed').waitFor();
  assert.equal((await canonical()).assigned,false);
  const refreshCalls=(await canonical()).assignmentCalls;
  await shot('committed-refresh-failed');
  await call('clear-read-failure');
  await test('recover').click();
  await poll(async()=>await test('toggle').isEnabled(),'read-only retry');
  assert.equal((await canonical()).assignmentCalls,refreshCalls);
  steps.push('Known commit with failed editor refresh keeps a reconciliation action; retry performs zero writes.');
  await call('assignment/CommittedUnconfirmed');
  await test('toggle').click();
  await poll(async()=> (await test('operation').innerText()).includes('unconfirmed'),'unconfirmed assignment');
  await shot('unconfirmed');
  const unknownCalls=(await canonical()).assignmentCalls;
  await select('Beta');
  await select('Alpha');
  assert.equal(await test('toggle').isDisabled(),true);
  await test('recover').click();
  await poll(async()=>await test('toggle').isEnabled(),'canonical verification');
  assert.equal((await canonical()).assignmentCalls,unknownCalls);
  assert.equal((await canonical()).assigned,true);
  steps.push('Unconfirmed assignment survives target reentry and resolves through canonical reads without replay.');
  await call('assignment/Conflict');
  await test('toggle').click();
  await test('recover').click();
  await test('adopt').waitFor();
  await shot('conflict-adoption');
  await test('adopt').click();
  await poll(async()=>await test('verify').isEnabled(),'adopt authoritative state');
  const proofs=(await canonical()).diagnosticCalls;
  await call('hold/proof');
  await test('verify').click();
  await poll(async()=> (await canonical()).held,'held inline proof');
  await shot('proof-pending');
  await call('change-capability');
  await call('release');
  await test('adopt').waitFor();
  assert.equal((await canonical()).proof,'NotRun');
  assert.equal((await canonical()).diagnosticCalls,proofs+1);
  await shot('proof-superseded');
  await test('adopt').click();
  await poll(async()=>await test('verify').isEnabled(),'proof adoption');
  await test('verify').click();
  await poll(async()=> (await canonical()).proof==='Verified','safe explicit inline proof');
  await poll(async()=>await test('verify').isEnabled(),'proof reconciliation');
  assert.equal((await canonical()).diagnosticCalls,proofs+2);
  steps.push('Changed capability inputs supersede a held inline diagnostic without proof publication; explicit later verification runs exactly once.');
  await call('hold/preview');
  await test('access-preview').click();
  await poll(async()=> (await canonical()).held,'held preview');
  await select('Beta');
  assert.equal((await canonical()).cancelled,true);
  await call('release');
  await page.waitForTimeout(200);
  assert.match(await heading.innerText(),/CAPA02 Beta/);
  assert.equal(await test('access-reason').inputValue(),'  Keep this raw rule  ');
  await shot('preview-superseded');
  await select('Alpha');
  await test('access-preview').click();
  await page.getByText('1 allowed',{exact:true}).waitFor();
  await shot('preview-ready');
  steps.push('Preview cancellation follows selection; late preview cannot replace B or erase the raw rule draft.');
  await test('details').click();
  await page.getByRole('dialog').waitFor();
  await shot('details');
  await page.keyboard.press('Escape');
  await page.getByRole('dialog').waitFor({state:'hidden'});
  for(const kind of ['tool','mcp','skill']) {
   await test('new-'+kind).click();
   await page.getByRole('dialog').waitFor();
   await shot('setup-'+kind);
   await page.keyboard.press('Escape');
   await page.getByRole('dialog').waitFor({state:'hidden'});
  }
  steps.push('Real details and tool, MCP and skill setup overlays render and close through owned references.');
  const lifecycle=await browser.newPage({viewport:{width:1600,height:1000}});
  lifecycle.on('pageerror',error=>errors.push(error.message));
  await lifecycle.goto(control+'/fixture/lifecycle');
  await lifecycle.getByTestId('agents-capability-search').waitFor({timeout:60000});
  await lifecycle.getByTestId('agents-capability-new-skill').click();
  await lifecycle.getByRole('dialog').waitFor();
  await call('unrelated');
  await lifecycle.getByTestId('fixture-owned-remove').waitFor();
  await lifecycle.getByTestId('fixture-owned-remove').click();
  await poll(async()=>await lifecycle.getByRole('dialog').count()===1,'only independent overlay');
  assert.equal(await lifecycle.getByTestId('agents-capability-search').count(),0);
  assert.match(await lifecycle.getByRole('dialog').innerText(),/Independent overlay/);
  await lifecycle.screenshot({path:path.join(out,'owner-disposed.jpeg'),type:'jpeg',quality:75});
  fs.writeFileSync(path.join(out,'owner-disposed-dom.txt'),await lifecycle.locator('body').innerText());
  await lifecycle.close();
  steps.push('A real non-navigation removal of the production panel closes its setup overlay while preserving the independent dialog.');
  assert.equal(state.curatorAvailable,true);
  await call('hold/curator');
  const chats=(await canonical()).curatorCalls;
  await test('curator-open').click();
  await poll(async()=> (await canonical()).held,'held curator launch');
  await select('Beta');
  await select('Alpha');
  assert.equal(await test('curator-open').isDisabled(),true);
  assert.equal((await canonical()).curatorCalls,chats+1);
  await shot('curator-pending');
  await call('release');
  await poll(async()=> !!(await canonical()).chatSessionId,'one created managed chat');
  assert.equal((await canonical()).curatorCalls,chats+1);
  await shot('curator-opened');
  steps.push('Curator remains single-flight across A -> B -> A and the real launcher creates one stable managed chat.');
  assert.deepEqual(errors,[]);
  fs.writeFileSync(path.join(out,'browser-summary.json'),JSON.stringify({status:'PASS',viewport:{width:1600,height:1000},browserVersion:browser.version(),steps,initial:state,final:await canonical(),pageErrors:errors},null,2));
 } catch(error) {
  await shot('failure');
  fs.writeFileSync(path.join(out,'browser-failure.json'),JSON.stringify({steps,error:error.message,pageErrors:errors},null,2));
  throw error;
 } finally {
  await call('release').catch(()=>{});
  await browser.close();
 }
})().catch(error=>{console.error(error);process.exitCode=1;});

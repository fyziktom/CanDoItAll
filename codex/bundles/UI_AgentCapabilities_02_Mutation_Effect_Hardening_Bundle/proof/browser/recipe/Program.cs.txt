internal static class BrowserEntry {
    public static async Task Main(string[] args) {
        var repository = Path.GetFullPath(args[^1]);
        await using var environment = CanDoItAllTestEnvironment.CreateUnder(Path.Combine(repository, ".mcp-state"), "capa02-browser-data");
        var database = environment.CreatePostgreSqlProfile("capa02-browser");
        var overrides = new Dictionary<string, string?> {
            ["Api:Authorization:Enabled"] = "false",
            ["ApiAccess:Authorization:Enabled"] = "false",
            ["WebHost:HttpsRedirectionEnabled"] = "false",
            ["Logging:LogLevel:Default"] = "Warning"
        };
        await using var services = await TestApplicationBootstrap.BuildServiceProviderAsync(
            database, "CanDoItAll.Web", TestSchemaBootstrapModules.Full, overrides);
        await using var scope = services.CreateAsyncScope();
        var workspace = scope.ServiceProvider.GetRequiredService<IAgentFrameworkWorkspaceService>();
        var capabilityId = await workspace.SaveCapabilityAsync(new CapabilityEditorModel {
            Kind = CapabilityKind.Skill,
            Key = "capa02-browser-inline",
            Name = "CAPA02 Inline inspection",
            Description = "Isolated in-place seam browser fixture.",
            EndpointOrPath = "inline:capa02-browser",
            ConfigurationJson = """{"inlineSkill":{"instructions":"Inspect the isolated fixture without external effects."}}""",
            Tags = ["capa02-fixture"]
        });
        var alphaId = await workspace.SaveAgentAsync(new AgentEditorModel {
            Name = "CAPA02 Alpha", RoleTitle = "Inspection", Instructions = "Inspect the isolated fixture.",
            Summary = "Read lifecycle fixture"
        });
        var betaId = await workspace.SaveAgentAsync(new AgentEditorModel {
            Name = "CAPA02 Beta", RoleTitle = "Inspection", Instructions = "Inspect the isolated fixture.",
            Summary = "Read lifecycle fixture"
        });
        var curator = (await workspace.ListAgentsAsync(false)).FirstOrDefault(CapabilityCuratorAgentIdentity.Matches);
        var delay = new FixtureControl { AlphaId = alphaId };
        var stop = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        var builder = WebApplication.CreateBuilder(new WebApplicationOptions { Args = [], ContentRootPath = Path.Combine(repository, ".mcp-state", "capa02-browser-fixture"), EnvironmentName = "Development" });
        builder.Configuration.AddInMemoryCollection(database.CreateConfigurationValues(overrides));
        TestApplicationBootstrap.ConfigureDefaultServices(builder.Services, builder.Configuration, builder.Environment, registerTestHostApplicationLifetime: false);
        builder.Services.AddRazorComponents().AddInteractiveServerComponents();
        builder.Services.AddAgentFrameworkUi();
        FixtureServices.Configure(builder.Services, delay);
        builder.WebHost.UseStaticWebAssets();
        builder.Logging.ClearProviders();
        builder.WebHost.UseUrls("http://127.0.0.1:17301");
        await using var control = builder.Build();
        control.UseAntiforgery();
        control.MapStaticAssets();
        control.MapRazorComponents<LifecycleDocument>().AddInteractiveServerRenderMode();
        control.MapPost("/fixture/unrelated", () => { _ = delay.OpenUnrelated?.Invoke(); return Results.Ok(); });
        control.MapPost("/fixture/assignment/{mode}", (AssignmentFault mode) => { delay.AssignmentMode = mode; return Results.Ok(); });
        control.MapPost("/fixture/hold/{kind}", (string kind) => { delay.Arm(kind); return Results.Ok(); });
        control.MapPost("/fixture/clear-read-failure", () => { delay.FailEditorRead = false; return Results.Ok(); });
        control.MapPost("/fixture/change-capability", async () => {
            var editor = await workspace.GetCapabilityEditorAsync(capabilityId);
            editor.Description = "Changed while a controlled inline diagnostic was held.";
            await workspace.SaveCapabilityAsync(editor);
            return Results.Ok();
        });
        control.MapGet("/fixture/state", async () => {
            var alpha = await workspace.GetAgentEditorAsync(alphaId);
            var capability = (await workspace.ListCapabilitiesAsync()).Single(item => item.Id == capabilityId);
            return Results.Json(new {
                alphaId, betaId, capabilityId,
                assigned = alpha.SelectedCapabilityIds.Contains(capabilityId), proof = capability.ProofStatus.ToString(),
                curatorAvailable = curator is { Status: AgentLifecycleStatus.Active, IsTemplate: false } && curator.Permissions.CanUseTools,
                delay.Held, delay.Cancelled, delay.AssignmentCalls, delay.DiagnosticCalls, delay.PreviewCalls,
                delay.CuratorCalls, delay.ChatSessionId, delay.FailEditorRead
            });
        });
        control.MapPost("/fixture/release", () => { delay.Release(); return Results.Ok(); });
        control.MapPost("/fixture/stop", () => { stop.TrySetResult(); return Results.Ok(); });
        await control.StartAsync();
        foreach (var variable in database.CreateEnvironmentVariables(overrides)) {
            Environment.SetEnvironmentVariable(variable.Key, variable.Value);
        }

        await using var application = new BrowserApplication(repository, delay);
        application.UseKestrel(5273);
        using var client = application.CreateClient(new WebApplicationFactoryClientOptions { AllowAutoRedirect = false });
        Console.WriteLine(JsonSerializer.Serialize(new { state = "FixtureReady", alphaId, betaId, capabilityId }));
        await stop.Task;
        delay.Release();
        await control.StopAsync();
    }
}

internal sealed class BrowserApplication(string repository, FixtureControl delay) : WebApplicationFactory<Program> {
    protected override void ConfigureWebHost(IWebHostBuilder builder) {
        builder.UseEnvironment("Development");
        builder.UseContentRoot(Path.Combine(repository, "src", "App", "CanDoItAll.Web"));
        builder.ConfigureLogging(logging => logging.SetMinimumLevel(LogLevel.Warning));
        builder.ConfigureServices(services => FixtureServices.Configure(services, delay));
    }
}

internal enum AssignmentFault { Normal, Rejected, Conflict, CommittedRefreshFailure, CommittedUnconfirmed }

internal sealed class FixtureControl {
    private readonly Lock gate = new();
    private TaskCompletionSource? held;
    private string? heldKind;
    private CancellationToken owner;
    public Guid AlphaId { get; init; }
    public Func<Task>? OpenUnrelated { get; set; }
    public AssignmentFault AssignmentMode { get; set; }
    public bool FailEditorRead { get; set; }
    public bool Held { get; private set; }
    public bool Cancelled => owner.IsCancellationRequested;
    public int AssignmentCalls;
    public int DiagnosticCalls;
    public int PreviewCalls;
    public int CuratorCalls;
    public Guid? ChatSessionId;
    public void Arm(string kind) {
        lock (gate) {
            heldKind = kind;
            held = new(TaskCreationOptions.RunContinuationsAsynchronously);
            Held = false;
        }
    }
    public Task PauseAsync(string kind, CancellationToken token) {
        lock (gate) {
            if (heldKind != kind) {
                return Task.CompletedTask;
            }
            heldKind = null;
            owner = token;
            Held = true;
            return held!.Task;
        }
    }
    public void Release() {
        lock (gate) {
            Held = false;
            held?.TrySetResult();
        }
    }
}

internal static class FixtureServices {
    public static void Configure(IServiceCollection services, FixtureControl control) {
        services.AddSingleton(control);
        services.AddScoped<IAgentCapabilitiesReads>(provider => new ObservedReads(
            new AgentCapabilitiesReads(provider.GetRequiredService<IAgentFrameworkWorkspaceService>()), control));
        services.AddScoped<IAgentCapabilityCommands>(provider => new ObservedCommands(
            new AgentCapabilityCommands(provider.GetRequiredService<IAgentFrameworkWorkspaceService>(),
                provider.GetRequiredService<ILogger<AgentCapabilityCommands>>()), control));
        services.AddSingleton<ICapabilityProofService>(provider => new ObservedProof(
            new CapabilityProofService(provider.GetRequiredService<IPhysicalFileSystemPathPolicyFactory>()), control));
        services.AddScoped<IAgentCapabilitySetupFlowService>(provider => new ObservedSetup(
            ActivatorUtilities.CreateInstance<AgentCapabilitySetupFlowService>(provider), control));
        services.AddScoped<IAgentChatLauncher>(provider => new ObservedLauncher(
            provider.GetRequiredService<AgentChatLauncherCompatibilityFacade>(), control));
    }
}

internal sealed class ObservedReads(IAgentCapabilitiesReads inner, FixtureControl control) : IAgentCapabilitiesReads {
    public Task<AgentCapabilitiesCatalog> LoadCatalogAsync(CancellationToken token) => inner.LoadCatalogAsync(token);
    public Task<AgentEditorModel> ReadEditorAsync(Guid agentId, CancellationToken token) => control.FailEditorRead && agentId == control.AlphaId
        ? Task.FromException<AgentEditorModel>(new IOException("Controlled post-commit read failure.")) : inner.ReadEditorAsync(agentId, token);
}
internal sealed class ObservedCommands(IAgentCapabilityCommands inner, FixtureControl control) : IAgentCapabilityCommands {
    public Task<CapabilityVerificationOutcome> DiagnoseAsync(Guid agentId, Guid capabilityId, CancellationToken token) => inner.DiagnoseAsync(agentId, capabilityId, token);
    public async Task<AgentCapabilityOperationStatus> AssignAsync(AgentCapabilityAssignmentAttempt attempt, CancellationToken token) {
        Interlocked.Increment(ref control.AssignmentCalls);
        var mode = control.AssignmentMode;
        control.AssignmentMode = AssignmentFault.Normal;
        await control.PauseAsync("assignment", token);
        if (mode == AssignmentFault.Rejected) {
            return AgentCapabilityOperationStatus.Rejected;
        }
        if (mode == AssignmentFault.Conflict) {
            return AgentCapabilityOperationStatus.Conflict;
        }
        var status = await inner.AssignAsync(attempt, token);
        if (status is AgentCapabilityOperationStatus.Committed or AgentCapabilityOperationStatus.CommittedWithWarning) {
            control.FailEditorRead = mode == AssignmentFault.CommittedRefreshFailure;
            return mode == AssignmentFault.CommittedUnconfirmed ? AgentCapabilityOperationStatus.Unconfirmed : status;
        }
        return status;
    }
}
internal sealed class ObservedProof(ICapabilityProofService inner, FixtureControl control) : ICapabilityProofService {
    public async Task<CapabilityVerificationResult> VerifyAsync(AgentDefinition agent, ProviderProfile? provider, CapabilityCatalogItem capability, CancellationToken token) {
        Interlocked.Increment(ref control.DiagnosticCalls);
        await control.PauseAsync("proof", token);
        return await inner.VerifyAsync(agent, provider, capability, token);
    }
}
internal sealed class ObservedSetup(IAgentCapabilitySetupFlowService inner, FixtureControl control) : IAgentCapabilitySetupFlowService {
    public Task<CapabilitySetupTestResult> TestToolSetupAsync(CapabilityToolSetupTestRequest request, CancellationToken token = default) => inner.TestToolSetupAsync(request, token);
    public Task<McpSetupTestResult> TestMcpSetupAsync(CapabilityMcpSetupTestRequest request, CancellationToken token = default) => inner.TestMcpSetupAsync(request, token);
    public async Task<CapabilityAccessPreviewResult> PreviewAccessAsync(CapabilityAccessPreviewRequest request, CancellationToken token = default) {
        Interlocked.Increment(ref control.PreviewCalls);
        var result = await inner.PreviewAccessAsync(request, token);
        await control.PauseAsync("preview", token);
        return result;
    }
}
internal sealed class ObservedLauncher(IAgentChatLauncher inner, FixtureControl control) : IAgentChatLauncher {
    public void ShowCatalog(AgentChatCatalogTab tab = AgentChatCatalogTab.Agents) => inner.ShowCatalog(tab);
    public Task<ActiveAgentChat> OpenChatAsync(Guid agentId, Guid chatSessionId, CancellationToken token = default) => inner.OpenChatAsync(agentId, chatSessionId, token);
    public async Task<ActiveAgentChat> StartNewChatAsync(Guid agentId, CancellationToken token = default) {
        Interlocked.Increment(ref control.CuratorCalls);
        await control.PauseAsync("curator", token);
        var chat = await inner.StartNewChatAsync(agentId, token);
        control.ChatSessionId = chat.ChatSessionId;
        return chat;
    }
}

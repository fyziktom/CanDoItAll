    [Fact]
    public async Task Waiting_project_gate_reads_the_committed_new_revision_before_replacement() {
        var probe = new Probe();
        await using var fixture = await Fixture.CreateAsync(probe);
        var original = await fixture.CreateAsync();
        await using var holder = await fixture.OwnerAsync();
        await using var transaction = await holder.Database.BeginTransactionAsync();
        await SerializableMutationScope.AcquireRelationalScopeLocksAsync(holder,
            [ProjectMutationScopeKeys.ForProject(fixture.Admission.ProjectId)], CancellationToken.None);
        probe.Armed = true;
        using var deadline = new CancellationTokenSource(TimeSpan.FromSeconds(10));
        var pending = fixture.Service.SaveAsync(fixture.Replace(original, "Stale waiter"), deadline.Token);
        try {
            var pid = await probe.WaitingPid.Task.WaitAsync(deadline.Token);
            var profile = fixture.Services.GetRequiredService<ICanonicalRuntimeDatabase>().Profile;
            await using var observer = new NpgsqlConnection(profile.ConnectionString);
            await observer.OpenAsync(deadline.Token);
            await using var query = new NpgsqlCommand("SELECT cardinality(pg_blocking_pids(@pid)) > 0", observer);
            query.Parameters.AddWithValue("pid", pid);
            while (!(bool)(await query.ExecuteScalarAsync(deadline.Token))!) {
                await Task.Delay(TimeSpan.FromMilliseconds(10), deadline.Token);
            }
            var row = await holder.Set<ProjectResource>().SingleAsync(resource => resource.Id == original.Expected.ResourceId, deadline.Token);
            row.Revision = Guid.NewGuid();
            row.Name = "Intervening retained metadata";
            await holder.SaveChangesAsync(deadline.Token);
            await transaction.CommitAsync(deadline.Token);
            Assert.Equal(ResourceMetadataRejectionReason.ResourceChanged,
                Assert.IsType<ResourceMetadataSaveOutcome.Rejected>(await pending).Reason);
            Assert.Equal(row.Revision, (await ResourceMetadataTestCommands.FoundAsync(fixture.Service, row.Id)).Expected.Revision.Value);
        } finally {
            await transaction.DisposeAsync();
            await pending;
        }
    }

    [Fact]
    public async Task Writer_persists_curated_configuration_and_reuses_exact_metadata_per_project_object() {
        await using var fixture = await Fixture.CreateAsync();
        var storageId = Guid.NewGuid();
        var config = new StorageObjectResourceConfig(ResourceFileSourceKey.ForStorage(storageId).Value, storageId,
            StorageProviderKind.FileSystem, StorageLocatorKind.RelativePath, "folder/file.txt", "file.txt", "text/plain", 42);
        var request = new StorageObjectResourceWriteRequest(fixture.Admission.ProjectId, "Stored file", ResourceSensitivity.Sensitive,
            config, fixture.Admission, await fixture.Service.ReadContextAsync(), Guid.NewGuid());
        var writer = fixture.Services.GetRequiredService<IStorageObjectResourceWriter>();
        var first = Assert.IsType<StorageObjectResourceWriteResult.Accepted>(await writer.SaveAsync(request));
        var again = Assert.IsType<StorageObjectResourceWriteResult.Accepted>(await writer.SaveAsync(request with { ResourceId = Guid.NewGuid(), Name = "Not a rename", Sensitivity = ResourceSensitivity.Normal }));
        Assert.True(first.Created);
        Assert.False(again.Created);
        Assert.Equal(first.Snapshot, again.Snapshot);
        await using var owner = await fixture.OwnerAsync();
        var row = Assert.Single(await owner.Set<ProjectResource>().AsNoTracking().ToArrayAsync());
        Assert.Equal(StorageObjectResourceConnectorPlugin.PluginKey, row.ConnectorPluginKey);
        Assert.Equal(StorageObjectResourceConnectorPlugin.SchemaVersion, row.ConfigSchemaVersion);
        Assert.Equal(ResourceValidationStatus.Valid, row.ValidationStatus);
        Assert.Equal(ResourceSensitivity.Sensitive, row.Sensitivity);
        Assert.Equal(StorageObjectResourceConnectorPlugin.Serialize(config), row.ConfigJson);
        Assert.DoesNotContain("folder/file.txt", row.LocationOrIdentifier, StringComparison.Ordinal);
        foreach (var forbidden in new[] { "handle", "route", "metadata" }) {
            Assert.DoesNotContain(forbidden, row.ConfigJson, StringComparison.OrdinalIgnoreCase);
        }
        await fixture.AssertSearchAsync(first.Snapshot);
        var ordinary = fixture.Replace(first.Snapshot, "Ordinary overwrite");
        Assert.Equal(ResourceMetadataRejectionReason.GovernedResource,
            Assert.IsType<ResourceMetadataSaveOutcome.Rejected>(await fixture.Service.SaveAsync(ordinary)).Reason);
    }

    private sealed class Fixture(TestApplication application, AsyncServiceScope scope, ProjectWriteAdmission admission) : IAsyncDisposable {
        public TestApplication Application => application;
        public IServiceProvider Services => scope.ServiceProvider;
        public ProjectWriteAdmission Admission => admission;
        public ResourcesService Service => Services.GetRequiredService<ResourcesService>();
        public static async Task<Fixture> CreateAsync(Probe? probe = null, Action<IServiceCollection>? configure = null) {
            var application = await TestApplication.CreateAsync(new() { ConfigureServices = services => {
                configure?.Invoke(services);
                probe?.Configure(services);
            } });
            var scope = application.Services.CreateAsyncScope();
            try {
                var admission = await OwnerPostcommitTestProbe.CreateProjectAsync(scope.ServiceProvider, "Resource original-state proof");
                return new(application, scope, admission);
            } catch {
                await scope.DisposeAsync();
                await application.DisposeAsync();
                throw;
            }
        }
        public ResourceEditorModel Editor(string name = "Original resource") => new() {
            ProjectId = admission.ProjectId, ExpectedProjectAdmission = admission, Name = name,
            ConnectorPluginKey = WebhookResourceConnectorPlugin.PluginKey, ConfigSchemaVersion = "1.0",
            ConfigJson = "{\"endpointUrl\":\"https://example.invalid/original\",\"method\":\"post\"}"
        };
        public Task<ReplaceResourceMetadataCommand> NewAsync() => ResourceMetadataTestCommands.NewAsync(Service, Editor());
        public async Task<ResourceMetadataSnapshot> CreateAsync()
            => Assert.IsType<ResourceMetadataSaveOutcome.Accepted>(await Service.SaveAsync(await NewAsync())).Snapshot;
        public ReplaceResourceMetadataCommand Replace(ResourceMetadataSnapshot original, string name)
            => new(original.Expected, admission, ResourceMetadataValues.Capture(Editor(name)));
        public Task<ResourcesDbContext> OwnerAsync() => Services.GetRequiredService<IDbContextFactory<ResourcesDbContext>>().CreateDbContextAsync();
        public Task<SearchDbContext> SearchAsync() => Services.GetRequiredService<IDbContextFactory<SearchDbContext>>().CreateDbContextAsync();
        public async Task AssertSearchAsync(ResourceMetadataSnapshot snapshot) {
            await using var search = await SearchAsync();
            var key = snapshot.Expected.ResourceId.ToString();
            var document = await search.Set<SearchDocument>().SingleAsync(row => row.SourceType == SearchDocument.ResourceSourceType && row.SourceKey == key);
            Assert.Equal(snapshot.Values.Name, document.Title);
            Assert.Equal(snapshot.Values.Description, document.Summary);
            Assert.Equal(Assert.IsType<ResourceProjectBinding.Bound>(snapshot.Expected.Project).ProjectId, document.ProjectId);
            Assert.Contains(snapshot.Values.LocationOrIdentifier, document.Body, StringComparison.Ordinal);
        }
        public async ValueTask DisposeAsync() {
            try {
                await scope.DisposeAsync();
            } finally {
                await application.DisposeAsync();
            }
        }
    }

    private sealed class FailingDiagnostics : ILogger<ResourceMetadataWriter> {
        public bool Armed { get; set; }
        public int Failures { get; private set; }
        public IDisposable? BeginScope<TState>(TState state) where TState : notnull => null;
        public bool IsEnabled(LogLevel logLevel) => true;
        public void Log<TState>(LogLevel logLevel, EventId eventId, TState state, Exception? exception, Func<TState, Exception?, string> formatter) {
            if (Armed) {
                Failures++;
                throw new IOException("The diagnostic sink failed after the committed operation.");
            }
        }
    }

    private enum Fault { None, LostAckAndCleanup, Cleanup }
    private sealed class Probe(Fault fault = Fault.None) {
        public bool Armed { get; set; }
        public bool PauseCreation { get; set; }
        public TaskCompletionSource CreationPaused { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public TaskCompletionSource ResumeCreation { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public TaskCompletionSource<int> WaitingPid { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public ConcurrentDictionary<Guid, byte> Mutations { get; } = new();
        public ConcurrentQueue<(string Sql, Guid[] Ids)> Updates { get; } = new();
        public ConcurrentQueue<string> Reads { get; } = new();
        public Fault Failure => fault;
        public int Commits;
        public int CleanupFaults;
        public void Configure(IServiceCollection services) => services.AddSingleton<IDbContextFactory<ResourcesDbContext>>(provider =>
            new Factory(new DbContextOptionsBuilder<ResourcesDbContext>(provider.GetRequiredService<DbContextOptions<ResourcesDbContext>>())
                .AddInterceptors(new SaveProbe(this), new CommitProbe(this), new DisposalProbe(this), new CommandProbe(this)).Options, this));
    }
    private sealed class Factory(DbContextOptions<ResourcesDbContext> options, Probe probe) : IDbContextFactory<ResourcesDbContext> {
        public ResourcesDbContext CreateDbContext() => new(options);
        public async Task<ResourcesDbContext> CreateDbContextAsync(CancellationToken cancellationToken = default) {
            if (probe.PauseCreation) {
                probe.PauseCreation = false;
                probe.CreationPaused.TrySetResult();
                await probe.ResumeCreation.Task.WaitAsync(cancellationToken);
            }
            return new(options);
        }
    }
    private sealed class SaveProbe(Probe probe) : SaveChangesInterceptor {
        public override ValueTask<InterceptionResult<int>> SavingChangesAsync(DbContextEventData eventData,
            InterceptionResult<int> result, CancellationToken cancellationToken = default) {
            if (probe.Armed && eventData.Context is { } context) {
                probe.Mutations.TryAdd(context.ContextId.InstanceId, 0);
            }
            return ValueTask.FromResult(result);
        }
    }
    private sealed class CommitProbe(Probe probe) : DbTransactionInterceptor {
        public override Task TransactionCommittedAsync(DbTransaction transaction, TransactionEndEventData eventData, CancellationToken cancellationToken = default) {
            if (eventData.Context is { } context && probe.Mutations.ContainsKey(context.ContextId.InstanceId)) {
                Interlocked.Increment(ref probe.Commits);
                if (probe.Failure == Fault.LostAckAndCleanup) {
                    throw new IOException("Lost acknowledgement after actual Resource commit.");
                }
            }
            return Task.CompletedTask;
        }
    }
    private sealed class DisposalProbe(Probe probe) : DbConnectionInterceptor {
        public override Task ConnectionDisposedAsync(DbConnection connection, ConnectionEndEventData eventData) {
            if (probe.Failure != Fault.None && eventData.Context is { } context && probe.Mutations.TryRemove(context.ContextId.InstanceId, out _)) {
                Interlocked.Increment(ref probe.CleanupFaults);
                throw new DbUpdateConcurrencyException("Cleanup cannot prove that the actual commit rolled back.");
            }
            return Task.CompletedTask;
        }
    }
    private sealed class CommandProbe(Probe probe) : DbCommandInterceptor {
        public override ValueTask<InterceptionResult<int>> NonQueryExecutingAsync(DbCommand command, CommandEventData eventData,
            InterceptionResult<int> result, CancellationToken cancellationToken = default) {
            if (probe.Armed && command.CommandText.Contains("pg_advisory_xact_lock", StringComparison.Ordinal) && command.Connection is NpgsqlConnection connection) {
                probe.WaitingPid.TrySetResult(connection.ProcessID);
            }
            return ValueTask.FromResult(result);
        }
        public override ValueTask<InterceptionResult<DbDataReader>> ReaderExecutingAsync(DbCommand command, CommandEventData eventData,
            InterceptionResult<DbDataReader> result, CancellationToken cancellationToken = default) {
            if (probe.Armed) {
                probe.Reads.Enqueue(command.CommandText);
            }
            if (probe.Armed && command.CommandText.Contains("UPDATE \"Resources_ProjectResources\"", StringComparison.Ordinal)) {
                probe.Updates.Enqueue((command.CommandText, command.Parameters.Cast<DbParameter>().Select(parameter => parameter.Value).OfType<Guid>().ToArray()));
            }
            return ValueTask.FromResult(result);
        }
    }
